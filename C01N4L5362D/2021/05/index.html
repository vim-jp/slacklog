  endfunction
else

    return ch_sendraw(a:server, json_encode([0, [a:method] + a:params]) . "\n")
  endfunction

  function! s:request(server, method, params) abort
    let [ok, err] = ch_evalexpr(a:server, [a:method] + a:params)
    if err isnot# v:null
      throw err
    endif
    return ok
  endfunction
