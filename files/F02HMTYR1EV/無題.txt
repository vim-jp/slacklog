let g:datafile = expand('~/.cache/vim/restore/data')

function! Layout2Data(wins) abort
  for win in a:wins
    if win[0] ==# 'leaf'
      let file = getbufinfo(getwininfo(win[1])[0]['bufnr'])[0]['name']
      let win[1] = {'file': filereadable(file) ? file : ''}
    else
      call Layout2Data(win[1])
    endif
  endfor
endfunction

function! WriteRestore() abort
  let wins = []

  let i = 1
  while empty(winlayout(i)) == 0
    let layout = [winlayout(i)]
    call Layout2Data(layout)

    call add(wins, layout)
    let i += 1
  endwhile

  execute 'redir! > ' . g:datafile
     silent! echo json_encode(wins) . "\n"
  redir END
endfunction

function! RestoreWins(wins) abort
  for win in a:wins
    if win[0] ==# 'row'
      let i = 1
      while i < len(win[1])
        if i >= 2
          execute "normal! \<C-w>p"
        endif
        execute 'rightbelow vsplit'
        call RestoreWins(win[1])

        let i = i + 1
      endwhile
    elseif win[0] ==# 'col'
      let i = 1
      while i < len(win[1])
        if i >= 2
          execute "normal! \<C-w>p"
        endif
        execute 'rightbelow split'
        call RestoreWins(win[1])

        let i = i + 1
      endwhile
    elseif win[0] ==# 'leaf'
      if win[1]['file'] !=# ''
        execute 'edit ' . win[1]['file']
      endif
    endif
  endfor
endfunction

function! ReadRestore() abort
  let lines = ''
  for line in readfile(g:datafile)
    let lines .= line
  endfor

  let data = json_decode(lines)
  let i = 1
  for tab_data in data
    call RestoreWins(tab_data)
    if len(data) > i
      execute 'tabedit'
    endif

    let i = i + 1
  endfor
endfunction
