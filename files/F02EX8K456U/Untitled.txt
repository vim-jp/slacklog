import { basename  } from "https://deno.land/std@0.106.0/path/mod.ts"

const results = await Deno.emit("./calc.ts", {
  compilerOptions: {
    declaration: true,
    declarationDir: ".",
  }
});

const files = results.files
Object.keys(files).map(async(v) => {
  const fileName = basename(v)
  await Deno.writeTextFile(fileName, files[v]);
})
