function! s:fern_settings() abort
  nnoremap <silent> <buffer> <nowait> p :<C-u>call Open_float_preview()<CR>
  autocmd BufLeave <buffer> call Close_preview_window()
endfunction

let s:win = v:false

function! Close_preview_window()
  if s:win
    try
      call nvim_win_close(s:win, v:true)
    catch
    endtry
  endif
endfunction

function! Centered_floating_window(file)
  call Close_preview_window()

  let width = min([&columns - 4, max([80, &columns - 80])])
  let height = min([&lines - 4, max([20, &lines - 40])])
  let top = ((&lines - height) / 2) - 1
  let left = (&columns - width) / 2
  let opts = {'relative': 'editor', 'row': top, 'col': left, 'width': width, 'height': height, 'style': 'minimal', 'border': 'single'}

  let l:buf = nvim_create_buf(v:false, v:true)
  let s:win = nvim_open_win(l:buf, v:false, opts)
  set winhl=Normal:Normal

  call win_execute(s:win, 'edit ' . a:file)
endfunction

function! Open_float_preview() abort
  let l:helper = fern#helper#new()
  if l:helper.sync.get_scheme() !=# 'file'
    return
  endif

  let l:path = l:helper.sync.get_cursor_node()['_path']
  call Centered_floating_window(l:path)
endfunction
