call wilder#set_option('pipeline', [
\   wilder#branch(
\     wilder#python_file_finder_pipeline({

\     wilder#cmdline_pipeline({'fuzzy': 1}),

\ ])

\ 'highlighter': wilder#basic_highlighter(),
\ 'highlights': {

\ }))

\ 'highlighter': wilder#basic_highlighter(),
\ 'separator': '  ',
\ 'left': [' '],
\ 'right': [' ', wilder#wildmenu_index()],
\ 'border': 'rounded',
\ })

\ 'accept_key': '<C-e>',
\ })

call wilder#set_option(

\  wilder#renderer_mux({
