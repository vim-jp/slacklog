import System'Context { Context }
import System'File
import System'array
import Core'Number

puts = \(ctx Context, str string) none {
  File'Write(ctx, File'StdOut(), str)
  File'Write(ctx, File'StdOut(), "\n")
  none
}

mod = \(a number, b number) number {
  if a >= b {
    mod(a - b, b)
  } else {
    a
  }
}

fizzbuzz = \(num number, acc [string]) [string] {
  if num < 0 {
    acc
  } else {
    m = mod(num, 15)
    fizzbuzz(num - 1,
      if m == 0 {
        [string "FizzBuzz", ...acc]
      } else if m == 3 | m == 6 | m == 9 {
        [string "Fizz", ...acc]
      } else if m == 5 | m == 10 {
        [string "Buzz", ...acc]
      } else {
        [string Number'String(num), ...acc]
      })
  }
}

printList = \(ctx Context, list [string]) none {
  if [x, ...xs] = list {
    puts(ctx, x())
    printList(ctx, xs)
  } else {
    none
  }
}

main = \(ctx Context) number {
  printList(ctx, fizzbuzz(100, [string]))
  0
}
