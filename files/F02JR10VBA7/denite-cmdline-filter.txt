" iキーでフィルタリング開始
autocmd FileType denite
      \ nnoremap <buffer><nowait> i <Cmd>call <SID>start_filter()<CR>

function! s:start_filter() abort
  augroup denite-cmdline-filter
    autocmd SafeState * call denite#call_map('filter', getcmdline()) | redraw
  augroup END

  " フィルタリング中のマッピング
  let filter_maps = {
        \   '<C-N>': '<Cmd>normal j<CR><Cmd>redraw<CR>',
        \   '<C-P>': '<Cmd>normal k<CR><Cmd>redraw<CR>',
        \   '<CR>':  '<Esc><CR>',
        \   '<Tab>': '<Esc><Tab>',
        \ }

  for [lhs, rhs] in items(filter_maps)
    execute 'cmap <silent><buffer><nowait>' lhs rhs
  endfor

  call input('', denite#get_status('input')[:-2])

  for lhs in keys(filter_maps)
    execute 'cunmap <buffer>' lhs
  endfor
  autocmd! denite-cmdline-filter
endfunction
