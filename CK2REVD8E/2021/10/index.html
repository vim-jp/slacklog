MOD = 998244353

n = int(input())
a = [0] + list(map(int, input().split()))
b = [0] + list(map(int, input().split()))

memo = [[0 for _ in range(3010)] for _ in range(3010)]

def solve(i, j):
    if memo[i][j]:
        return memo[i][j]
    elif i <= n-1:
        memo[i][j] = sum([solve(i+1, elm) for elm in range(max(j, a[i+1]), b[i+1]+1)]) % MOD
        return memo[i][j]
    else:
        return 1
