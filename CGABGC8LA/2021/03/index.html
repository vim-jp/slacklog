func decodePsd(this js.Value, args []js.Value) interface{} {
	img, _, err := image.Decode(bytes.NewReader(psd))
	if err != nil {
		panic(err)
	}
	out := new(bytes.Buffer)
	err = png.Encode(out, img)
	if err != nil {
		panic(err)
	}
	enc := base64.StdEncoding.EncodeToString(out.Bytes())
	return js.ValueOf(enc)
}
