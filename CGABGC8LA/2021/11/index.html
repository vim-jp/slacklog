This adds two new snippets, centred around string manipulation.

The first snippet applies to variables of type string, and is called
split. When used, it replaces someStringVar.split! with
strings.Split(someStringVar, "|"), where | indicates the location of
the cursor.

The second snippet is essentially the same as the first, but in
"reverse". Meaning that rather than going from a string to a slice of
strings, it goes from a slice of strings to a single string:
someStringSlice.join! -> strings.Join(someStringSlice, "|").

（cgoでQuickJSを呼び出せていい感じにmoduleが使える）

package main

import (
	"<http://github.com/eihigh/qjs|github.com/eihigh/qjs>"
	"<http://github.com/eihigh/qjs/stdlib/qjslog|github.com/eihigh/qjs/stdlib/qjslog>"
)

func main() {
	if err := _main(); err != nil {
		panic(err)
	}
}

func _main() error {
	rt, err := qjs.NewRuntime()
	if err != nil {
		return err
	}
	defer rt.Free()

	// use packages
	pkgs := []qjs.Package{
		{Name: "log", Init: qjslog.Init},
	}

	ctx, err := rt.NewContext(pkgs, nil)
	if err != nil {
		return err
	}
	defer ctx.Free()

	// export the value
	v, err := ctx.Eval("export const a = 42;", "foobar", qjs.EVAL_TYPE_MODULE|qjs.EVAL_FLAG_STRICT)
	if err != nil {
		return err
	}
	defer v.Free()

	// import the value and the package
	v, err = ctx.Eval(
		"import {a} from 'foobar'; import {log} from 'go'; log.Print(`${a}`);",
		"",
		qjs.EVAL_TYPE_MODULE|qjs.EVAL_FLAG_STRICT,
	)
	if err != nil {
		return err
	}
	defer v.Free()

	return nil
}
