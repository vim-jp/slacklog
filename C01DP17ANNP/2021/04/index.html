カーソル下にMarkdownの画像 ( ![...](...) )が置かれている場合、そのURLを抽出するScript

function! Fuga()
  let [l:line, l:col] = searchpairpos('!\[[^\]]*\](', '', ')', 'bcn')
  return matchstrpos(getline(l:line), '\(^!\[[^\]]*\](\)\@<=\([^ )]\+\)\(\( \+=\d\+x\d*\)\?)\)\@=', l:col-1)
endfunction
