nnoremap <expr> <Plug>(input-refeedkeys) <SID>refeed()
function! s:refeed() abort
  let l:keys = ''
  while getchar(1)
    let l:keys ..= nr2char(getchar())
  endwhile
  call timer_start(0, { -> feedkeys(l:keys, 'i') })
  echomsg 'You was automatically typed: ' .. input('')
endfunction

                Characters in {string} are queued for processing as if they
