cocのアクションが関数でしか呼べないの面倒だなーーって思っていたらコマンドで呼びやすくするやつができました
補完もちょっと効く！
function! s:complete_coc(argLead, cmdLine, cursorPos) abort
  if match(a:cmdLine[0 : a:cursorPos], '\<Coc!\?\s\+\a*$') < 0
    return ''
  endif

  if empty(get(s:, 'coc_actions', ''))
    let s:coc_actions = join(map(getcompletion("CocAction('", 'help'), 'v:val[11:-3]'), "\n")
  endif

  return s:coc_actions
endfunction

command! -nargs=+ -complete=custom,s:complete_coc -bang Coc
      \ call CocAction{<bang>0 ? '' : 'Async'}(<f-args>)

つかいかた：
:Coc rename           " => :call CocActionAsync('rename')
:Coc doHover float    " => :call CocActionAsync('doHover', 'float')
:Coc! format          " => :call CocAction('format')

ご注意：
• 追加の引数は文字列しか渡せないよ
• 呼んでも意味ないアクションも補完に出てくるよ
• cocのhelpのタグが変わったら補完効かなくなるかも
気が向いたらプラグインにするかも

" vvvvvvvvvvvvvvvvvvvvvvvvvvvvvvvvvvvvvvvvvvvvvvvvvvvvvvvvvvvvvvvvvvvvvvvvvvv
" Automatic reload from disk

" Triger `autoread` when files changes on disk
" <https://unix.stackexchange.com/questions/149209/refresh-changed-content-of-file-opened-in-vim/383044#383044>
" <https://vi.stackexchange.com/questions/13692/prevent-focusgained-autocmd-running-in-command-line-editing-mode>
    autocmd FocusGained,BufEnter,CursorHold,CursorHoldI *
            \ if mode() !~ '\v(c|r.?|!|t)' && getcmdwintype() == '' | checktime | endif

" Notification after file change
" <https://vi.stackexchange.com/questions/13091/autocmd-event-for-autoread>
autocmd FileChangedShellPost *
  \ echohl WarningMsg | echo "File changed on disk. Buffer reloaded." | echohl None
