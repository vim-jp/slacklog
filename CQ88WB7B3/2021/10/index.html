cocのアクションが関数でしか呼べないの面倒だなーーって思っていたらコマンドで呼びやすくするやつができました
補完もちょっと効く！
function! s:complete_coc(argLead, cmdLine, cursorPos) abort
  if match(a:cmdLine[0 : a:cursorPos], '\<Coc!\?\s\+\a*$') < 0
    return ''
  endif

  if empty(get(s:, 'coc_actions', ''))
    let s:coc_actions = join(map(getcompletion("CocAction('", 'help'), 'v:val[11:-3]'), "\n")
  endif

  return s:coc_actions
endfunction

command! -nargs=+ -complete=custom,s:complete_coc -bang Coc
      \ call CocAction{<bang>0 ? '' : 'Async'}(<f-args>)

つかいかた：
:Coc rename           " => :call CocActionAsync('rename')
:Coc doHover float    " => :call CocActionAsync('doHover', 'float')
:Coc! format          " => :call CocAction('format')

ご注意：
• 追加の引数は文字列しか渡せないよ
• 呼んでも意味ないアクションも補完に出てくるよ
• cocのhelpのタグが変わったら補完効かなくなるかも
気が向いたらプラグインにするかも
