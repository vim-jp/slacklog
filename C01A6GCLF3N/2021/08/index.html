例えばこういう簡単な描写があるとします。
#include <ncurses.h>

int main()
{
	initscr();
	start_color();

	init_color(1, 255, 255, 255);
	init_color(2, 1, 1, 1);
	init_pair(1, 1, 2);

	attron(COLOR_PAIR(1));
	addstr("aaa");
	attroff(COLOR_PAIR(1));

	refresh();
	getch();
	endwin();

	return 0;
}

	addstr("aaa");

	init_color(1, 11, 11, 11);
