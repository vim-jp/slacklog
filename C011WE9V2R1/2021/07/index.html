@echo off

setlocal

set PATH=C:\Program Files\Docker\Docker\resources;%PATH%

set ARGS=
for /f "usebackq tokens=1 delims==" %%i in (`docker ps -a -q`) do call :add %%i
if "%ARGS%" neq "" (
  docker rm -f %ARGS%
)

set ARGS=
for /f "usebackq tokens=1 delims==" %%i in (`docker images -q`) do call :add %%i
if "%ARGS%" neq "" (
  docker rmi -f %ARGS%
)
exit /b

:add
if "%ARGS%"=="" (
  set ARGS=%*
) else (
  set ARGS=%ARGS% %*
)

docker の image を消すバッチとして使ってる。
