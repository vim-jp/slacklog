  call s:assert.equals(WebDevIconsGetFileTypeSymbol(a:filename), a:icon)
endfunction

function! s:suite.__OneArgument_VimIcon__()
  let targetfilenames = ['.vimrc', 'vimrc', '.gvimrc', '_gvimrc', 'test.vim']
  let expecticon = ''
  let child = themis#suite('OneArgument_VimIcon')

  for targetfilename in targetfilenames
    let child[targetfilename] = funcref('s:Assert', [targetfilename, expecticon])
  endfor
endfunction

この include のやつ、文字列の場合は部分文字列を探すけど、配列の場合は要素を探す。利便性とかを考えると正しいとは思うんだけど、どうも一貫性がないような気もしてしまう。
どういうことかと言うと、 include('abcde', 'bcd') とあったら、abcde は bcd の並びを含んでいるので true になる。
では include(['a', 'b', 'c', 'd', 'e'], ['b', 'c', 'd']) はどうかと言うと、左の配列は ['b', 'c', 'd'] と言う配列を要素に含んでいないので false になる。
実際問題として「配列の一部の並びを含んでいるかどうか」なんてのをチェックしたいシーンはそうそうない。ないんだが…一貫性が気になってしまうw
