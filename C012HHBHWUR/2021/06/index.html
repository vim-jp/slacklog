Sub Query または Materialized View を使った例。
CREATE MATERIALIZED VIEW commented_articles
AS
SELECT
	a.id AS article_id,
	a.updated_at AS article_updated_at,
	MAX(c.updated_at) AS latest_commented_at
FROM articles AS a
LEFT OUTER JOIN comments AS c
ON c.article_id = a.id
GROUP BY a.id;

CREATE INDEX commented_articles_latest_index ON commented_articles(
	latest_commented_at DESC NULLS FIRST,
	article_updated_at DESC
);

SET enable_seqscan TO 'off';

-- use SUBQUERY
EXPLAIN
SELECT
	s.id,
	MAX(s.updated_at) AS t
FROM (
	SELECT
		a.id,
		c.updated_at
	FROM articles AS a
	LEFT OUTER JOIN comments AS c
	ON c.article_id = a.id
	ORDER BY c.updated_at DESC NULLS FIRST
) AS s
GROUP BY s.id
LIMIT 10
OFFSET 11;

-- use MATERIALIZED VIEW
EXPLAIN
SELECT
	a.id,
	c.latest_commented_at
FROM articles AS a
INNER JOIN commented_articles AS c
ON c.article_id = a.id
ORDER BY c.latest_commented_at DESC NULLS FIRST
LIMIT 10
OFFSET 11;
