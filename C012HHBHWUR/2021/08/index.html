create policy select_policy on data.shops for select to api
using (
  ((request.user_role() = 'norole' or request.user_role() = 'webuser') and
    id in (select shop_id from api.actors a inner join api.roles r on a.role_id = r.id where a.user_id = request.user_id()))
  or
  (request.user_role() = 'webuser' and
    has_scope('USER', request.actor_id(), 'READ', 'shops') and
    id = actor_shop(request.actor_id()))
  or
  (request.user_role() = 'webuser' and
    has_scope('SHOP', request.actor_id(), 'READ', 'shops') and
    id = actor_shop(request.actor_id()))
  or
  (request.user_role() = 'webuser' and
    has_scope('CLIENT', request.actor_id(), 'READ', 'shops') and
    client_id = actor_client(request.actor_id()))
  or
  (request.user_role() = 'webuser' and
    has_scope('ALL', request.actor_id(), 'READ', 'shops'))
);
