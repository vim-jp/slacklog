function pyset(){
    local TARGET="$(pyenv install --list | fzf | sed -E 's/^\s*(.*)\s*$/\1/')"
    pyenv install --skip-existing "$TARGET"
    pyenv ${1:-local} "$TARGET"
}

<https://github.com/tidymodels/workflows/issues/111|#111 `predict.workflow` fails if recipe step includes `step_select`>
*The problem*

If model is fit and predict on workflows::workflow(), predict.workflow fails if recipe step includes step_select.
However, the same recipe works fine without using workflows package.

*Reproducible example*


library(magrittr)

preprocessor <- recipes::recipe(mtcars, mpg ~ .) %>%
  recipes::step_select(mpg, wt)
  
model <- parsnip::linear_reg() %>%
  parsnip::set_engine("lm")

# with workflows
workflows::workflow() %>%
  workflows::add_recipe(preprocessor) %>%
  workflows::add_model(model) %>%
  generics::fit(mtcars) %>%
  predict(mtcars)
#> Error: Can't subset columns that don't exist.
#> x Column `mpg` doesn't exist.

# without workflows
input <- preprocessor %>%
  recipes::prep() %>%
  recipes::juice()
model %>%
  generics::fit(mpg ~ ., data = input) %>%
  predict(input)
