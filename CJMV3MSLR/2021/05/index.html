短縮入力では、入力されて短縮入力展開のトリガーとなった文字が |v:char| にセット

マッピングで何も実行させない場合は、評価結果が空文字列になるようにします。も

のなら (例えば、画面の更新など)、"\<Ignore>" を返すようにします。

        func s:OpenPopup()
          call popup_create(... arguments ...)
          return "\<Ignore>"
        endfunc

こせてしまいます。そのため、次のものは制限されます:
- バッファのテキストの変更 |textlock|。
- 他のバッファの編集。
- |:normal| コマンド。
- カーソルの移動は可能ですが、後で元に戻されます。

  inoremap <expr> <C-L> nr2char(getchar())
  inoremap <expr> <C-L>x "foo"

function! s:exec_command() abort
  let in = input(':')
  if empty(in)
    return
  endif
  echom ''
  redraw
  let parts = split(in, ' ')
  let cmd = ''
  if len(parts) == 1
    let cmd = in
  else
    let cmd = parts[0]
  endif

  if cmd[0] is# '!' || cmd[:-1] is# '!' || exists(':' .. cmd)
    exe in
  elseif executable(cmd)
    call s:term_start(in)
  endif
endfunction
