ジェネレータ (プログラミング)
ジェネレータは、プログラムにおいて、数列の各要素の値などを次々と生成（ジェネレート）し他の手続きに渡す、という機能を持っている手続きである。値を渡す方法としては、コールバックのようにして他の手続きを呼ぶものもあれば、呼び出される度に次々と異なる値を返す関数であることもある。

(define (map-tree proc sexp)
  (if (pair? sexp)
      (cons (map-tree proc (car sexp)) (map-tree proc (cdr sexp)))
      (proc sexp)))

(define-er-macro define-generator (form rename cmp?)
  (match form
    [(_ (fn . args) . body)
     (quasirename rename
       (define ,fn
         (let* ((retval #f)
                (eval-rem #f)
                (yield
                  (lambda v
                    (let/cc c
                      (set! eval-rem c)
                      (retval (apply values v)))))
                (proc (lambda ,args ,@(map-tree
                                        (lambda (atom) (if (eq? atom 'yield) (rename atom) atom))
                                        body))))
            (lambda ,args

                  ((procedure? eval-rem)
                   (eval-rem))
                  (else (eof-object))))))))]))
