#!/bin/bash
#
#   This file echoes four gradients with 24-bit color codes
#   to the terminal to demonstrate their functionality.
#   The foreground escape sequence is ^[38;2;<r>;<g>;<b>m
#   The background escape sequence is ^[48;2;<r>;<g>;<b>m
#   <r> <g> <b> range from 0 to 255 inclusive.
#   The escape sequence ^[0m returns output to default

SEPARATOR=':'

setBackgroundColor()
{
    echo -en "\x1b[48${SEPARATOR}2${SEPARATOR}$1${SEPARATOR}$2${SEPARATOR}$3""m"
}

resetOutput()
{
    echo -en "\x1b[0m\n"
}

# Gives a color $1/255 % along HSV
# Who knows what happens when $1 is outside 0-255
# Echoes "$red $green $blue" where
# $red $green and $blue are integers
# ranging between 0 and 255 inclusive
rainbowColor()
{ 
    let h=$1/43
    let f=$1-43*$h
    let t=$f*255/43
    let q=255-t

    if [ $h -eq 0 ]
    then
        echo "255 $t 0"
    elif [ $h -eq 1 ]
    then
        echo "$q 255 0"
    elif [ $h -eq 2 ]
    then
        echo "0 255 $t"
    elif [ $h -eq 3 ]
    then
        echo "0 $q 255"
    elif [ $h -eq 4 ]
    then
        echo "$t 0 255"
    elif [ $h -eq 5 ]
    then
        echo "255 0 $q"
    else
        # execution should never reach here
        echo "0 0 0"
    fi
}

for i in `seq 0 127`; do
    setBackgroundColor $i 0 0
    echo -en " "
done
resetOutput
for i in `seq 255 128`; do
    setBackgroundColor $i 0 0
    echo -en " "
done
resetOutput

for i in `seq 0 127`; do
    setBackgroundColor 0 $i 0
    echo -n " "
done
resetOutput
for i in `seq 255 128`; do
    setBackgroundColor 0 $i 0
    echo -n " "
done
resetOutput

for i in `seq 0 127`; do
    setBackgroundColor 0 0 $i
    echo -n " "
done
resetOutput
for i in `seq 255 128`; do
    setBackgroundColor 0 0 $i
    echo -n " "
done
resetOutput

for i in `seq 0 127`; do
    setBackgroundColor `rainbowColor $i`
    echo -n " "
done
resetOutput
for i in `seq 255 128`; do
    setBackgroundColor `rainbowColor $i`
    echo -n " "
done
resetOutput

<https://github.com/fish-shell/fish-shell/issues/3190|#3190 Redirecting with tee>
Today I wanted to run some program several time storing one particular line of output of the program to a file.

It was relatively easy with something like this:


$ for x in (seq 10)
      echo $x
      myprogram | grep "Total time" >> stats.txt
end


But since the program ran for fairly long time, I wanted to be able to see the output.

in bash, I would have called the tee like this:


      myprogram | tee >(grep "Total time" >> stats.txt)


But fish shell complained:

> Invalid redirection target: (grep "Total time" >> stats.txt)

It is possible I am doing something wrong, but at least in case of bash, the same redirection seems to be working like a charm...

Here's a minimal test case:


> for x in (seq 500); echo $x >> source.txt; end
> cat source.txt | tee >(grep 450 > 450.txt)
Invalid redirection target: (grep 450 > 450.txt)
fish: cat source.txt | tee >(grep 450 > 450.txt)
                           ^


I would have expected this snippet to print out all 500 lines from source.txt to stdout and written a file 450.txt with one line containing 450 in it.
