拙作ghf コマンドでは、 gh repo list | fzf みたいなことを gh repoに限らずサブサブコマンドに list を持つものすべてでできるようにするため、
1. gh repo list の標準出力を変数に代入し標準エラーを一時ファイルにリダイレクト

3. なければ続行

func DataDir() string {
	var path string
	if a := os.Getenv(XDG_DATA_HOME); a != "" {
		path = filepath.Join(a, "gh")
	} else if b := os.Getenv(LOCAL_APP_DATA); runtime.GOOS == "windows" && b != "" {
		path = filepath.Join(b, "GitHub CLI")
	} else {
		c, _ := os.UserHomeDir()
		path = filepath.Join(c, ".local", "share", "gh")
	}

	return path
}
