function git_auto_save() {
  if [[ -d .git ]] && [[ -f .git/auto-save ]] && [[ $(find .git/auto-save -mmin -$((60)) | wc -l) -eq 0 ]]; then
    if [[ ! -f ".git/MERGE_HEAD" ]] && [[ $(git --no-pager diff --cached | wc -l) -eq 0 ]] && [[ ! -f .git/index.lock ]] && [[ ! -d .git/rebase-merge ]] && [[ ! -d .git/rebase-apply ]]; then
      touch .git/auto-save && git add --all && git commit --no-verify --message "Auto save: $(date -R)" >/dev/null && git reset HEAD^ >/dev/null
      echo "Git auto save!"
    fi
  fi
}

自分がやってる(まだkeep-baseははいってない)やつ
default-branch = "!basename \"$(git symbolic-ref --short refs/remotes/origin/HEAD)\""
branch-root = !git merge-base $(git default-branch) HEAD
fixing = "!f () {\
	if [ \"${#}\" -eq 1 ]; then\
		GIT_SEQUENCE_EDITOR=':' git rebase -i ${*};\
	else\
		GIT_SEQUENCE_EDITOR=':' git rebase -i $(git branch-root);\
	fi;\
};f"

master/mainを default-branch で取り、それから分岐元を branch-root で取り、それをもって POSIX shellで処理のパスになる ":" をeditorにして、 git rebase -i する、ということをしています。
