package main

import (
	"fmt"
	"os"
	"os/exec"
	"path/filepath"
	"strings"
	"sync"
)

func main() {
	cmd := exec.Command("git", "ls-files")
	b, err := cmd.Output()
	if err != nil {
		panic(err)
	}
	files := strings.Split(string(b), "\n")

	limit := 100
	sem := make(chan struct{}, limit)
	var wg sync.WaitGroup
	for _, f := range files {
		if f == "" {
			continue
		}
		wg.Add(1)
		f := f
		sem <- struct{}{}
		go func() {
			defer wg.Done()
			cmd := exec.Command("git", "log", "-n", "1", "--format=%ad", "--date=unix", filepath.Clean(f))
			cmd.Stderr = os.Stderr
			b, err := cmd.Output()
			if err != nil {
				panic(err)
			}
			<-sem
			fmt.Println(f, strings.TrimSpace(string(b)))
		}()
	}
	wg.Wait()
}
